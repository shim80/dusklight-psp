cmake_minimum_required(VERSION 3.25)
project(dusklight_getawait_heart_probe LANGUAGES C CXX)
set(DUSKLIGHT_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../dusklight-main")
add_subdirectory("${DUSKLIGHT_SOURCE_ROOT}/platforms/psp" "${CMAKE_BINARY_DIR}/psp-runtime")
add_executable(getawait_heart_probe main.cpp)
target_compile_features(getawait_heart_probe PRIVATE cxx_std_20)
target_compile_options(getawait_heart_probe PRIVATE -O2 -G0 -Wall -Wextra -Werror -fsigned-char -fno-exceptions -fno-rtti)
target_link_libraries(getawait_heart_probe PRIVATE dusk_psp_renderer dusk_psp_resources dusk_psp_platform pspgum pspgu pspge pspdisplay pspdebug pspctrl)
create_pbp_file(TARGET getawait_heart_probe TITLE "Dusklight GETAWAIT Heart Probe" OUTPUT_DIR "${CMAKE_BINARY_DIR}")
